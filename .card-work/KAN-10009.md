        self.status = "Open"

    def validate_architecture_scope(self) -> bool:
        """Verifica se o perfil exige conhecimento formal em Arquitetura de Sistemas."""
        if "Architecture" in self.domains and "System Design" in self.required_skills.get("skills", []):
            return True
        return False
