from abc import ABC, abstractmethod
from typing import Dict, Any

class IProductArchitecture(ABC):
    """
    Interface abstrata para definir os contratos de Arquitetura de Produto.
    Esta classe define as regras e restrições que qualquer implementação de produto deve seguir.
    """
    @abstractmethod
    def define_architecture_constraints(self) -> Dict[str, Any]:
        """Define as restrições de arquitetura e os limites de escopo do produto."""
        pass

    @abstractmethod
    def validate_feature_feasibility(self, feature_request: str) -> bool:
        """Valida a viabilidade técnica e de negócio de uma nova feature proposta."""
