class ServiceRegistry:
    """
    Gerencia o registro de todos os serviços e suas dependências no sistema.
    """
    def __init__(self):
        # Dicionário principal para mapear nomes de serviços a suas implementações
        self._services = {}
        # Novo atributo para rastreamento de escopo de produto e arquitetura
        self.product_scope_tags = {} 

    def register_service(self, service_name: str, implementation_path: str, dependencies: list):
        """Registra um novo serviço no sistema."""
        self._services[service_name] = {
            "path": implementation_path,
            "dependencies": dependencies
        }
