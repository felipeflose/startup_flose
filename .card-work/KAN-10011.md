def prioritize_features(backlog: list) -> list:
    """Prioriza funcionalidades utilizando uma matriz de priorização ponderada (incluindo auditorias anti-alucinação)."""
    
    # Definição de pesos para a priorização (Exemplo de regra de negócio)
    PRIORITY_WEIGHTS = {
        "anti_hallucination_audit": 0.95,  # Alta prioridade de segurança/qualidade
        "core_feature_A": 0.70,
        "feature_B": 0.50
    }
    
    def calculate_score(feature):
        base_score = feature.complexity_score
        
        # Adiciona o peso específico para auditorias de segurança
        if feature.name.startswith("anti_hallucination"):
            base_score += PRIORITY_WEIGHTS.get(feature.name, 0.1)
            
        return base_score

    prioritized = sorted(backlog, key=calculate_score, reverse=True)
    return prioritized
