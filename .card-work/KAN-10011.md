from .models import Task, Priority

def get_pending_tasks(user_id: int) -> list[Task]:
    """
    Busca tarefas pendentes, aplicando a regra de priorização baseada no nível de negócio.
    Esta função suporta a priorização complexa exigida pelo Product Owner.
    """
    pending_tasks = db.query.filter(Task.status == 'pending', Task.user_id == user_id).all()

    # Aplicação da lógica de priorização (Exemplo: Prioridade alta para auditorias)
    prioritized_tasks = []
    for task in pending_tasks:
        if task.type in ['auditoria_anti_alucinação', 'arquitetura_crítica']:
            task.priority = Priority.HIGH
        else:
            task.priority = Priority.MEDIUM
        prioritized_tasks.append(task)
