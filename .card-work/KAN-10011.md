class Feature:
    """
    Representa uma funcionalidade de produto ou um grande requisito de negócio.
    """
    def __init__(self, feature_id: str, title: str, description: str, area: str, priority: str, dependencies: list = None):
        self.feature_id = feature_id
        self.title = title
        self.description = description
        self.area = area
        self.priority = priority  # Ex: 'High', 'Medium', 'Low'

        self.status = "Backlog"
